# name=Trap Metal Kit
"""Trap Metal Kit: ONE FL Studio script, ONE MIDI input, everything on it.

FL Studio only runs one controller script per MIDI input, and this kit has four (DJ decks, vocal mixer setup,
plugin probe, pink theme). This file holds all four, so you assign it once and get the lot.

THIS FILE IS GENERATED by tools/build_fl_bundle.py from the four device_*.py files beside it. Edit the CONFIG block
just below to change keys / turn parts off. To change anything deeper, edit the source script and rebuild (a rebuild
resets this CONFIG block). Build id: @@BUILD@@

Default key map (FL names note 60 "C5"; every number here is a MIDI note):

    36-67   launch Performance Mode clips: 4 decks (playlist tracks 1-4) x 8 clips       (C3 - G5)
    68-71   stop deck 1-4                                                                (G#5 - B5)
    72      PROBE   list every plugin in the mixer and the parameters FL can automate    (C6)  read-only
    73      AUDIT   check Nectar 4 / Ozone 12 / a distortion are on the right inserts    (C#6) read-only
    74      SETUP   build the vocal mixer (names, colours, routing, sends, arm the lead) (D6)  press twice
    75      REPAINT paint channels, mixer, playlist and patterns pink / purple / red     (D#6) press twice
    mod wheel (CC 1)  crossfades deck 1 against deck 2 on mixer inserts 10 and 11

SETUP and REPAINT change your project, so they only run if you press the key a second time within a few seconds.
Everything it does is reported in View > Script output (and under the controller's tab in Options > MIDI Settings).

Install: python3 tools/install_fl_studio.py   then see docs/00-fl-studio-quickstart.md.
Tested against stubs of the FL modules and checked against Image-Line's published API stubs; not run inside FL Studio.
"""
import time
import traceback
import types

# ======================================================================================================== CONFIG
# Turn a part off by setting it to False.
ENABLED = {
    "TrapDJ": True,        # keys 36-71 launch clips; mod wheel crossfades
    "VocalSetup": True,    # SETUP / AUDIT keys
    "PluginProbe": True,   # PROBE key
    "PinkTheme": True,     # pink painting at start-up + REPAINT key
}

# Change any setting of any part: "Part": {"SETTING": value}. A setting must exist in that part's source script
# (see its top-of-file constants); a typo is reported in Script output and ignored.
OVERRIDES = {
    # Decks sit on mixer inserts 10 and 11 so the mod-wheel crossfader never touches the vocal inserts (1-8).
    "TrapDJ": {"DECK_A_MIXER": 10, "DECK_B_MIXER": 11},
    # The four command keys sit directly above the DJ block, inside what a 49/61/88-key keyboard can play.
    "PluginProbe": {"PROBE_NOTE": 72},
    "VocalSetup": {"AUDIT_NOTE": 73, "SETUP_NOTE": 74},
    "PinkTheme": {"REPAINT_NOTE": 75},      # add "PAINT_ON_START": False to stop it painting when FL starts
}

# SETUP and REPAINT only run on the second press within this many seconds. 0 = run on the first press.
CONFIRM_SECONDS = 3.0
# ================================================================================================ end of CONFIG

SECTION_ORDER = ["TrapDJ", "VocalSetup", "PluginProbe", "PinkTheme"]
GUARDED = {   # part -> (setting holding the key, what pressing it does)
    "VocalSetup": ("SETUP_NOTE", "rebuild mixer inserts 1-8 as the vocal layout"),
    "PinkTheme": ("REPAINT_NOTE", "repaint every channel, mixer insert, playlist track and pattern"),
}

@@SECTIONS@@

_modules = {}      # part name -> loaded module (only enabled parts that loaded)
_pending = {}      # guarded note -> time of its first press
_reported = set()  # (part, callback) pairs already reported as failing, so OnIdle cannot flood the output


def _log(message):
    print("Trap Metal Kit: " + message)


def _load_all():
    _modules.clear()
    for name in ENABLED:
        if name not in SECTION_ORDER:
            _log("ENABLED has an unknown part '%s' (known: %s); ignored" % (name, ", ".join(SECTION_ORDER)))
    for name in OVERRIDES:
        if name not in SECTION_ORDER:
            _log("OVERRIDES has an unknown part '%s' (known: %s); ignored" % (name, ", ".join(SECTION_ORDER)))
    for name in SECTION_ORDER:
        if not ENABLED.get(name, True):
            continue
        filename = "device_%s.py" % name
        try:
            module = types.ModuleType("device_" + name)
            module.__file__ = filename
            exec(compile(_SOURCES[name], filename, "exec"), module.__dict__)
            for key, value in OVERRIDES.get(name, {}).items():
                if hasattr(module, key):
                    setattr(module, key, value)
                else:
                    _log("OVERRIDES: %s has no setting called '%s' (typo?); ignored" % (name, key))
            _modules[name] = module
        except Exception:
            _log("%s could not be loaded and is switched off:\n%s" % (name, traceback.format_exc()))


def _call(callback, *args):
    """Run `callback` on every loaded part that defines it, stopping once one marks the event handled."""
    for name in SECTION_ORDER:
        module = _modules.get(name)
        handler = getattr(module, callback, None) if module is not None else None
        if handler is None:
            continue
        try:
            handler(*args)
        except Exception:
            if (name, callback) not in _reported:
                _reported.add((name, callback))
                _log("%s.%s failed (the other parts keep working; reported once):\n%s"
                     % (name, callback, traceback.format_exc()))
            continue
        if args and getattr(args[0], "handled", False):
            return True
    return False


def claims():
    """Which notes and mixer inserts each loaded part uses: ({note: [who, ...]}, {insert: [who, ...]})."""
    notes, inserts = {}, {}

    def claim(table, key, who):
        table.setdefault(key, []).append(who)

    dj = _modules.get("TrapDJ")
    if dj is not None:
        for note in range(dj.FIRST_NOTE, dj.stop_first_note() + dj.ROWS):
            claim(notes, note, "TrapDJ")
        if dj.XFADE_CC is not None:
            for insert in (dj.DECK_A_MIXER, dj.DECK_B_MIXER):
                claim(inserts, insert, "TrapDJ crossfader")
    vocal = _modules.get("VocalSetup")
    if vocal is not None:
        claim(notes, vocal.SETUP_NOTE, "VocalSetup SETUP")
        claim(notes, vocal.AUDIT_NOTE, "VocalSetup AUDIT")
        for insert in range(vocal.FIRST_INSERT, vocal.FIRST_INSERT + len(vocal.LAYOUT)):
            claim(inserts, insert, "VocalSetup layout")
    probe = _modules.get("PluginProbe")
    if probe is not None:
        claim(notes, probe.PROBE_NOTE, "PluginProbe PROBE")
    theme = _modules.get("PinkTheme")
    if theme is not None and theme.REPAINT_NOTE is not None:
        claim(notes, theme.REPAINT_NOTE, "PinkTheme REPAINT")
    return notes, inserts


def find_conflicts():
    """Human-readable lines for any note or mixer insert that two parts both want."""
    notes, inserts = claims()
    lines = []
    for note, who in sorted(notes.items()):
        if len(who) > 1:
            lines.append("note %d is claimed by %s" % (note, " and ".join(who)))
    for insert, who in sorted(inserts.items()):
        if len(who) > 1:
            lines.append("mixer insert %d is used by %s" % (insert, " and ".join(who)))
    return lines


def _command_notes():
    notes = set()
    for name, (setting, _) in GUARDED.items():
        module = _modules.get(name)
        if module is not None and getattr(module, setting, None) is not None:
            notes.add(getattr(module, setting))
    for name, setting in (("VocalSetup", "AUDIT_NOTE"), ("PluginProbe", "PROBE_NOTE")):
        module = _modules.get(name)
        if module is not None:
            notes.add(getattr(module, setting))
    return notes


def _guard(event):
    """True if `event` is a guarded key pressed for the first time: swallow it and wait for the second press."""
    if not CONFIRM_SECONDS or event.data2 <= 0:
        return False
    for name, (setting, what) in GUARDED.items():
        module = _modules.get(name)
        if module is None or getattr(module, setting, None) != event.data1:
            continue
        now = time.time()
        first = _pending.get(event.data1)
        if first is not None and now - first <= CONFIRM_SECONDS:
            del _pending[event.data1]
            return False
        _pending[event.data1] = now
        _log("press note %d again within %g seconds to %s" % (event.data1, CONFIRM_SECONDS, what))
        event.handled = True
        return True
    return False


def describe():
    """The active key map as lines of text."""
    lines = []
    dj = _modules.get("TrapDJ")
    if dj is not None:
        stop = dj.stop_first_note()
        line = "notes %d-%d launch clips (%d decks x %d), %d-%d stop decks" % (
            dj.FIRST_NOTE, stop - 1, dj.ROWS, dj.COLUMNS, stop, stop + dj.ROWS - 1)
        if dj.XFADE_CC is not None:
            line += "; CC %d crossfades mixer inserts %d and %d" % (dj.XFADE_CC, dj.DECK_A_MIXER, dj.DECK_B_MIXER)
        lines.append(line)
    probe = _modules.get("PluginProbe")
    if probe is not None:
        lines.append("note %d lists every plugin and parameter" % probe.PROBE_NOTE)
    vocal = _modules.get("VocalSetup")
    if vocal is not None:
        lines.append("note %d audits the vocal chain; note %d builds the vocal mixer" % (vocal.AUDIT_NOTE, vocal.SETUP_NOTE))
    theme = _modules.get("PinkTheme")
    if theme is not None and theme.REPAINT_NOTE is not None:
        lines.append("note %d repaints everything pink" % theme.REPAINT_NOTE)
    return lines


_load_all()


def OnInit():
    _pending.clear()
    _reported.clear()
    _log("ready (build %s). Parts on: %s" % (BUILD, ", ".join(n for n in SECTION_ORDER if n in _modules) or "none"))
    for line in describe():
        _log("  " + line)
    if CONFIRM_SECONDS:
        _log("  keys that change your project need a second press within %g s" % CONFIRM_SECONDS)
    for line in find_conflicts():
        _log("WARNING: " + line + " - change one in CONFIG so they do not fight")
    _call("OnInit")


def OnDeInit():
    _call("OnDeInit")


def OnIdle():
    _call("OnIdle")


def OnNoteOn(event):
    if _guard(event):
        return
    _call("OnNoteOn", event)


def OnNoteOff(event):
    _call("OnNoteOff", event)
    if not event.handled and event.data1 in _command_notes():
        event.handled = True    # the release of a command key should not reach the channel rack either


def OnControlChange(event):
    _call("OnControlChange", event)
